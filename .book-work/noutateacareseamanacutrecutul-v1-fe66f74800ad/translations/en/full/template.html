"""Exemplu didactic; nu evalueaza niciun LLM."""
from itertools import product
from math import ceil, log
def value(node, bits):
    """Evalueaza un arbore validat, fara eval()."""
    if type(node) is int:
        if not 0 <= node < len(bits):
            raise ValueError("Indice de variabila invalid")
        return bits[node]
    if not isinstance(node, tuple) or not node:
        raise ValueError("Nod invalid")
    op, *args = node
    arity = {"not": 1, "and": 2, "or": 2, "xor": 2}
    if not isinstance(op, str) or op not in arity:
        raise ValueError("Operator invalid")
    if len(args) != arity[op]:
        raise ValueError("Operator sau aritate invalida")
    values = [value(arg, bits) for arg in args]
    if op == "not":
        return 1 - values[0]
    a, b = values
    if op == "and":
        return a & b
    if op == "or":
        return a | b
    return a ^ b
def table(node, n):
    if type(n) is not int or not 1 <= n <= 16:
        raise ValueError("n trebuie sa fie intre 1 si 16")
    return tuple(value(node, bits)
                 for bits in product((0, 1), repeat=n))
def compare(candidate, library, n):
    """Noutate functionala fata de o biblioteca finita."""
    target = table(candidate, n)
    matches = [name for name, node in library.items()
               if table(node, n) == target]
    return not matches, matches
def all_inputs_true(n):
    node = 0
    for index in range(1, n):
        node = ("and", node, index)
    return node
def main():
    library = {"AND": ("and", 0, 1),
               "OR": ("or", 0, 1),
               "XOR": ("xor", 0, 1)}
    old = ("not", ("or", ("not", 0), ("not", 1)))
    new = ("not", ("xor", 0, 1))
    assert compare(old, library, 2) == (False, ["AND"])
    assert compare(new, library, 2) == (True, [])
    print("Parafraza AND: veche")
    print("XNOR fata de biblioteca: noua")
    n = 8
    baseline = ("xor", 0, 1)
    candidate = ("xor", baseline, all_inputs_true(n))
    first, second = table(baseline, n), table(candidate, n)
    witnesses = [bits for bits, a, b in
                 zip(product((0, 1), repeat=n),
                     first, second) if a != b]
    assert witnesses == [(1,) * n]
    assert first[:-1] == second[:-1]
    assert first != second
    print("Martor unic:", witnesses[0])
    print("Coliziune dupa eliminarea martorului: True")
    miss = (1 - 2 ** (-n)) ** 100
    needed = ceil(log(0.05) / log(1 - 2 ** (-n)))
    print(f"Ratare dupa 100 observatii: {miss:.6f}")
    print("Observatii pentru ratare <= 0.05:", needed)
    print("Toate verificarile exacte au trecut.")
if __name__ == "__main__":
    main()
